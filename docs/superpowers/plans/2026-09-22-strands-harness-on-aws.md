# strands-harness-on-aws Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A cloneable reference repo that runs the Strands harness (`create_harness()`) as a read-only AWS Ops Investigator, locally and on Amazon Bedrock AgentCore Runtime (HTTP + A2A runtimes, AgentCore Memory, Cognito JWT auth, CloudWatch observability), with a standalone `doctor.py` and a pitfalls field guide.

**Architecture:** One `build_agent()` factory wraps `create_harness()`; `platform.py` switches sessions/memory/skills/tool profile by env (`local` vs `cloud`). Two thin entrypoints (`main.py` for AgentCore HTTP/SSE, `a2a_main.py` for A2A 1.0) share it. A deterministic `ReadOnlyGuard` intervention plus a (service, operation) allowlist in the AWS tools plus an IAM deny block make the agent read-only in three layers. Deployment is an arm64 container via the npm `@aws/agentcore` CLI (CDK) driven by `agentcore.json`.

**Tech Stack:** Python 3.12 (image) / 3.10+ (supported); `strands-harness==0.1.1`, `strands-agents==1.56.0`, `bedrock-agentcore==1.23.1`, `a2a-sdk[http-server]>=1.1,<2`, `boto3`, `aws-opentelemetry-distro~=0.18.0`; `uv`; `pytest` + `botocore.stub.Stubber`; `ruff`; npm `@aws/agentcore@0.30.x`; Docker buildx (arm64); AWS account 123456789012 / profile `<your-profile>` / region `us-west-2` for the owner's smoke test.

**Spec:** `docs/superpowers/specs/2026-09-22-strands-harness-on-aws-design.md` (v2). Read it first; task rationale and the pitfall numbers (P1–P27) come from it.

```python
"""Where am I running, and which backends should the harness use?

`cloud` profile = inside AgentCore Runtime (the CDK stack injects MEMORY_OPSINVESTIGATORMEMORY_ID).
`local` profile = a laptop. See spec P2, P5, P6, P21, P23.
"""

from __future__ import annotations

import os
import shutil
import tempfile
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path
from typing import Literal

MEMORY_ENV = "MEMORY_OPSINVESTIGATORMEMORY_ID"
DEFAULT_MODEL = "bedrock/global.anthropic.claude-sonnet-5"
DEFAULT_EFFORT = "medium"
EFFORTS = ("auto", "off", "minimal", "low", "medium", "high", "xhigh", "max")
APP_DIR = Path(__file__).resolve().parent.parent  # app/ops_investigator
AWS_DOCS_MCP_URL = "https://knowledge-mcp.global.api.aws"


class RuntimeConfigError(RuntimeError):
    """Unrecoverable configuration problem; the message carries the fix."""


@dataclass(frozen=True)
class Runtime:
    profile: Literal["local", "cloud"]
    region: str | None
    memory_id: str | None
    workspace: Path
    app_dir: Path
    model: str
    effort: str
    session_id: str
    actor_id: str
    mcp_servers: dict | None


def detect(session_id: str, actor_id: str, env: Mapping[str, str] | None = None) -> Runtime:
    env = os.environ if env is None else env
    memory_id = env.get(MEMORY_ENV) or None
    region = env.get("AWS_REGION") or env.get("AWS_DEFAULT_REGION") or None
    profile: Literal["local", "cloud"] = "cloud" if memory_id else "local"
    if profile == "cloud" and not region:
        raise RuntimeConfigError(
            "AWS_REGION is not set inside the runtime. Fix: add {name: AWS_REGION, value: <region>} to the runtime's "
            "envVars in agentcore/agentcore.json (P5)."
        )
    effort = env.get("HARNESS_EFFORT", DEFAULT_EFFORT)
    if effort not in EFFORTS:
        raise RuntimeConfigError(f"HARNESS_EFFORT={effort!r} is not one of {EFFORTS}.")
    workspace = Path(tempfile.gettempdir()) / "ops-workspace" if profile == "cloud" else APP_DIR
    mcp_servers = None
    if env.get("ENABLE_AWS_DOCS_MCP") == "1":
        mcp_servers = {"aws-docs": {"url": AWS_DOCS_MCP_URL, "transport": "streamable-http", "continue_on_error": True}}
    return Runtime(
        profile=profile,
        region=region,
        memory_id=memory_id,
        workspace=workspace,
        app_dir=APP_DIR,
        model=env.get("HARNESS_MODEL", DEFAULT_MODEL),
        effort=effort,
        session_id=session_id,
        actor_id=actor_id,
        mcp_servers=mcp_servers,
    )


def enter_workspace(rt: Runtime) -> Path:
    """Make the harness's cwd writable (P2). In cloud, /app is read-only for the non-root user.

    The harness `environment` plugin reads AGENTS.md from cwd, so copy it along.
    """
    if rt.profile == "local":
        return rt.workspace
    rt.workspace.mkdir(parents=True, exist_ok=True)
    src = rt.app_dir / "AGENTS.md"
    if src.exists():
        shutil.copyfile(src, rt.workspace / "AGENTS.md")
    os.chdir(rt.workspace)
    return rt.workspace
```

```python
"""Actor identity. Cloud: JWT `sub` from the allowlisted Authorization header (spec 4.4, P24).

The AgentCore CUSTOM_JWT authorizer validated the token before the container saw it; we only decode.
"""

from __future__ import annotations

import base64
import json
import re
from collections.abc import Mapping

ACTOR_RE = re.compile(r"^[A-Za-z0-9_-]{1,64}$")


class ActorError(ValueError):
    pass


def _header(headers: Mapping[str, str], name: str) -> str | None:
    for k, v in headers.items():
        if k.lower() == name.lower():
            return v
    return None


def _jwt_claims(token: str) -> dict:
    parts = token.split(".")
    if len(parts) != 3:
        raise ActorError("Authorization bearer token is not a JWT (expected three dot-separated segments).")
    payload = parts[1] + "=" * (-len(parts[1]) % 4)
    try:
        return json.loads(base64.urlsafe_b64decode(payload.encode()))
    except (ValueError, UnicodeDecodeError) as e:
        raise ActorError("Authorization bearer token payload is not valid base64url JSON.") from e


def _validate(actor: str) -> str:
    if not ACTOR_RE.match(actor or ""):
        raise ActorError("actor id must match ^[A-Za-z0-9_-]{1,64}$ (memory namespaces are built from it).")
    return actor


def actor_from_headers(headers: Mapping[str, str], *, profile: str, payload_actor: str | None = None) -> str:
    if profile == "cloud":
        raw = _header(headers, "Authorization")
        if not raw or not raw.lower().startswith("bearer "):
            raise ActorError(
                "Missing Authorization: Bearer <jwt>. The runtime is configured for CUSTOM_JWT; get a token with "
                "`make token` (scripts/token.py) and pass it on every invoke."
            )
        return _validate(str(_jwt_claims(raw.split(" ", 1)[1].strip()).get("sub", "")))
    hdr = _header(headers, "X-Actor-Id")
    return _validate(hdr or payload_actor or "local-dev")
```

```python
"""Per-invocation API budget kept in the agent's invocation_state (spec: reviewer M5)."""

from __future__ import annotations


def take(invocation_state: dict | None, key: str, limit: int) -> bool:
    """Consume one unit; return False when the budget for `key` is exhausted."""
    if invocation_state is None:
        return True
    budget = invocation_state.setdefault("budget", {})
    used = budget.get(key, 0)
    if used >= limit:
        return False
    budget[key] = used + 1
    return True
```

```python
"""Read-only AWS API tool. Three layers keep it read-only: this allowlist, ReadOnlyGuard, and IAM (spec 4.6, P26)."""

from typing import Any

from botocore.exceptions import BotoCoreError, ClientError

OUTPUT_CAP = 40_000
CALL_BUDGET = 40
READ_PREFIXES = ("describe_", "list_", "get_", "lookup_", "search_", "query_")

# Operations that are "reads" by name but return secret material or transcripts (spec 4.6). Mirrors the IAM deny.
DENY: frozenset[tuple[str, str]] = frozenset({

    ("dynamodb", "execute_statement"), ("dynamodb", "batch_execute_statement"),
    ("lambda", "get_function"), ("lambda", "get_function_configuration"), ("lambda", "get_layer_version"),
    ("ec2", "get_password_data"), ("ec2", "get_console_output"), ("ec2", "get_console_screenshot"),
    ("ec2", "describe_instance_attribute"),
    ("cloudformation", "get_template"),
    ("ecs", "describe_task_definition"),
    ("codebuild", "batch_get_projects"),
    ("apigateway", "get_api_keys"), ("apigateway", "get_api_key"),
    ("cognito-idp", "admin_get_user"), ("cognito-idp", "list_users"),
    ("sqs", "receive_message"),
    ("kinesis", "get_records"), ("kinesis", "get_shard_iterator"),
    ("sts", "assume_role"), ("sts", "get_federation_token"), ("sts", "get_session_token"),
    ("logs", "get_log_events"), ("logs", "filter_log_events"), ("logs", "start_query"), ("logs", "get_query_results"),
})


def is_read_operation(service: str, operation: str) -> bool:
    op = operation.strip().lower()
    return op.startswith(READ_PREFIXES) and (service.strip().lower(), op) not in DENY


def _hint_for(code: str) -> str:
    if code in ("AccessDenied", "AccessDeniedException", "UnauthorizedOperation"):
        return "The execution role lacks this permission or it is in the explicit deny list (policies/execution-role-deny.json)."
    if code in ("Throttling", "ThrottlingException", "TooManyRequestsException"):
        return "Throttled by AWS; narrow the query or wait before retrying."
    return "Check the service/operation name (boto3 snake_case) and parameters."


def _cap(payload: Any) -> tuple[Any, bool]:
    text = json.dumps(payload, default=str)
    if len(text) <= OUTPUT_CAP:
        return payload, False
    return text[:OUTPUT_CAP] + "...", True


@tool(context=True)
def aws_call(
    service: str,
    operation: str,
    parameters: dict | None = None,
    region: str | None = None,
    query: str | None = None,
    max_items: int = 50,
    tool_context: ToolContext | None = None,
) -> str:
    """Call a read-only AWS API (boto3 snake_case operation, e.g. service="ec2", operation="describe_instances").

    Only Describe*/List*/Get*/Lookup*/Search*/Query* operations are allowed, minus operations that return secrets.
    Use `query` (JMESPath) to shrink large responses and `max_items` to bound pagination. Output is capped at 40k chars.
    """
    state = tool_context.invocation_state if tool_context is not None else None
    if not is_read_operation(service, operation):
        return json.dumps({"error": f"{service}:{operation} is not permitted",
                           "hint": "read-only investigator: Describe*/List*/Get*/Lookup*/Search*/Query* only, "
                                   "and never secret-bearing reads (S3 objects, SSM parameters, DynamoDB items, ...)."})
    if not _budget.take(state, "aws_call", CALL_BUDGET):
        return json.dumps({"error": "budget_exhausted", "hint": f"{CALL_BUDGET} AWS calls per turn; summarise what you have."})
    ids = (state or {}).get("ids", {})
    try:
        client = _clients.make_client(service, region, session_id=ids.get("session_id", ""), actor_id=ids.get("actor_id", ""))
        params = dict(parameters or {})
        if client.can_paginate(operation):
            pages = client.get_paginator(operation).paginate(**params, PaginationConfig={"MaxItems": max_items})
            data: Any = pages.build_full_result()
        else:
            data = getattr(client, operation)(**params)
        if isinstance(data, dict):
            data.pop("ResponseMetadata", None)
        if query:
            data = jmespath.search(query, data)
        capped, truncated = _cap(data)
        return json.dumps({"service": service, "operation": operation, "region": region, "data": capped,
                           "truncated": truncated, "returned": len(data) if isinstance(data, (list, dict)) else 1},
                          default=str)
    except ClientError as e:
        code = e.response.get("Error", {}).get("Code", "ClientError")
        return json.dumps({"error": f"{code}: {e.response.get('Error', {}).get('Message', '')}"[:500], "hint": _hint_for(code)})
    except (BotoCoreError, ValueError, TypeError, AttributeError) as e:
        return json.dumps({"error": f"{type(e).__name__}: {e}"[:500], "hint": _hint_for("")})
```

```python
"""CloudWatch Logs Insights, bounded. The agent's own log groups are never readable (spec 4.6)."""

from __future__ import annotations

import json
import os
import time
from collections.abc import Sequence

from botocore.exceptions import BotoCoreError, ClientError
from strands import tool
from strands.types.tools import ToolContext

from ops_investigator.tools import _budget, _clients

DEFAULT_PREFIXES = ("/aws/lambda/", "/aws/ecs/", "/aws/apigateway/", "/aws/eks/")
FORBIDDEN_PREFIXES = ("/aws/bedrock-agentcore/", "aws/spans")
CALL_BUDGET = 10
MAX_MINUTES = 24 * 60
POLL_SECONDS = 1.0
POLL_LIMIT = 25


def configured_prefixes() -> tuple[str, ...]:
    raw = os.environ.get("OPS_LOG_GROUP_PREFIXES", "")
    return tuple(p.strip() for p in raw.split(",") if p.strip()) or DEFAULT_PREFIXES


def allowed_log_group(name: str, prefixes: Sequence[str]) -> bool:
    if any(name.startswith(f) for f in FORBIDDEN_PREFIXES):
        return False
    return any(name.startswith(p) for p in prefixes)


@tool(context=True)
def logs_insights(
    log_group: str,
    query_string: str,
    minutes: int = 60,
    limit: int = 100,
    region: str | None = None,
    tool_context: ToolContext | None = None,
) -> str:
    """Run a CloudWatch Logs Insights query over the last `minutes` on one allowed log group and return rows.

    Allowed groups are configured by OPS_LOG_GROUP_PREFIXES. Keep queries narrow (filter + limit).
    """
    state = tool_context.invocation_state if tool_context is not None else None
    if not allowed_log_group(log_group, configured_prefixes()):
        return json.dumps({"error": f"log group {log_group} is not in the allowed prefixes",
                           "hint": f"allowed prefixes: {', '.join(configured_prefixes())}"})
    if not _budget.take(state, "logs_insights", CALL_BUDGET):
        return json.dumps({"error": "budget_exhausted", "hint": f"{CALL_BUDGET} Logs Insights queries per turn"})
    minutes = max(1, min(int(minutes), MAX_MINUTES))
    limit = max(1, min(int(limit), 500))
    ids = (state or {}).get("ids", {})
    try:
        client = _clients.make_client("logs", region, session_id=ids.get("session_id", ""), actor_id=ids.get("actor_id", ""))
        end = int(time.time())
        qid = client.start_query(logGroupName=log_group, startTime=end - minutes * 60, endTime=end,
                                 queryString=query_string, limit=limit)["queryId"]
        for _ in range(POLL_LIMIT):
            res = client.get_query_results(queryId=qid)
            if res.get("status") in ("Complete", "Failed", "Cancelled", "Timeout"):
                rows = [{c["field"]: c["value"] for c in row} for row in res.get("results", [])]
                return json.dumps({"log_group": log_group, "status": res["status"], "rows": rows,
                                   "stats": res.get("statistics", {})}, default=str)
            time.sleep(POLL_SECONDS)
        return json.dumps({"error": "query still running", "hint": "narrow the time window or query", "query_id": qid})
    except ClientError as e:
        code = e.response.get("Error", {}).get("Code", "ClientError")
        return json.dumps({"error": f"{code}: {e.response.get('Error', {}).get('Message', '')}"[:500],
                           "hint": "check the log group name and the execution role's logs permissions"})
    except BotoCoreError as e:
        return json.dumps({"error": f"{type(e).__name__}: {e}"[:500], "hint": "transient AWS error; retry once"})
```

```python
"""Cost Explorer summary. The API lives in us-east-1 and bills ~$0.01 per request, hence the small budget."""

import json

def cost_explorer(days: int = 7, group_by: str = "SERVICE", region: str | None = None,
                  tool_context: ToolContext | None = None) -> str:
    """Unblended cost for the last `days` days grouped by SERVICE, REGION, LINKED_ACCOUNT, or USAGE_TYPE (top 25)."""

    if not _budget.take(state, "cost_explorer", CALL_BUDGET):

                                        Granularity="MONTHLY" if days > 31 else "DAILY", Metrics=["UnblendedCost"],
                                        GroupBy=[{"Type": "DIMENSION", "Key": group_by}])

        groups = sorted(({"key": k, "usd": round(v, 2)} for k, v in totals.items()), key=lambda x: -x["usd"])[:25]
        return json.dumps({"start": start.isoformat(), "end": end.isoformat(), "group_by": group_by,
                           "total_usd": round(sum(totals.values()), 2), "groups": groups})
    except ClientError as e:
        code = e.response.get("Error", {}).get("Code", "ClientError")
        return json.dumps({"error": f"{code}: {e.response.get('Error', {}).get('Message', '')}"[:500],
```

```python
"""Deterministic read-only intervention (spec P10, 6.1).

A policy *string* passed to create_harness(interventions=...) becomes an LLM classifier that pauses with an
interrupt; in a headless runtime nobody resumes it and the session wedges. This handler returns Deny, never Confirm.
"""

from __future__ import annotations

import logging
import re
from typing import Any

from strands.hooks import BeforeToolCallEvent
from strands.interventions import InterventionHandler
from strands.interventions.actions import Deny, Proceed

from ops_investigator.tools.aws_call import is_read_operation

log = logging.getLogger("ops_investigator.guard")

DANGEROUS_TOOLS: frozenset[str] = frozenset({"shell", "write", "edit", "web_fetch", "web_search"})
# UX-level check for the local profile only; the enforced control for AWS mutation is IAM + aws_call's allowlist.
MUTATING_AWS_CLI = re.compile(
    r"\baws\b.*\b(create|put|delete|update|modify|terminate|run-instances|start|stop|reboot|attach|detach|"
    r"authorize|revoke|add|remove|associate|disassociate|enable|disable|invoke|send|publish|rm|cp|sync|mv|"
    r"assume-role|get-secret-value|get-parameter)\b",
    re.IGNORECASE,
)


class ReadOnlyGuard(InterventionHandler):
    def __init__(self, denied_tools: frozenset[str] = DANGEROUS_TOOLS) -> None:
        self.denied_tools = denied_tools

    def _deny(self, name: str, reason: str) -> Deny:
        log.warning("guard.denied tool=%s reason=%s", name, reason)
        return Deny(reason=reason)

    def before_tool_call(self, event: BeforeToolCallEvent, **kwargs: Any) -> Proceed | Deny:
        tool_use = event.tool_use
        name = str(tool_use.get("name", ""))
        inp = tool_use.get("input") or {}
        if name in self.denied_tools:
            return self._deny(name, f"{name} is disabled for the read-only investigator in this profile")
        if name == "aws_call":
            service, op = str(inp.get("service", "")), str(inp.get("operation", ""))
            if not is_read_operation(service, op):
                return self._deny(name, f"read-only: {service}:{op} is not a permitted read operation")
        if name == "shell":
            cmd = str(inp.get("command", ""))
            if MUTATING_AWS_CLI.search(cmd):
                return self._deny(name, "read-only: mutating or secret-reading aws CLI call refused")
        return Proceed()
```

```python
import functools
from collections.abc import Mapping

from strands.types.agent import Limits

BACKGROUND_TASKS = {"agentic": [], "max_concurrency": 2, "timeout": 300}

def _cap_stream(agent: Agent, limits: Limits) -> Agent:

    limits, so bind the cap here (the Agent constructor has no `limits` keyword)."""

        memory=False,

        context_manager="auto",
        name="reader",
```

```python
"""Versioned SSE frames (spec 4.4). Never put tool bodies or prompts on the wire beyond text deltas."""

from __future__ import annotations

from typing import Any

SCHEMA_VERSION = 1

# USD per 1M tokens: (input, output, cache_read). Verify against the Bedrock price list before release (spec 10.5).
PRICES = {
    "sonnet": (3.0, 15.0, 0.30),
    "opus": (15.0, 75.0, 1.50),
    "haiku": (1.0, 5.0, 0.10),
}


def frame(type_: str, **fields: Any) -> dict:
    return {"v": SCHEMA_VERSION, "type": type_, **fields}


def _content_bytes(content: list | None) -> int:
    total = 0
    for block in content or []:
        if isinstance(block, dict) and "text" in block:
            total += len(str(block["text"]))
        elif isinstance(block, dict) and "json" in block:
            total += len(str(block["json"]))
    return total


def map_event(event: Any) -> dict | None:
    if not isinstance(event, dict):
        return None
    if "data" in event and isinstance(event.get("data"), str):
        return frame("text", delta=event["data"])
    if event.get("reasoning") and event.get("reasoningText"):
        return frame("thinking", delta=event["reasoningText"])
    if event.get("type") == "tool_use_stream":
        tu = event.get("current_tool_use") or {}
        inp = tu.get("input") if isinstance(tu.get("input"), dict) else {}
        return frame("tool_use", name=tu.get("name", ""), input_keys=sorted(inp.keys()))
    if event.get("type") == "tool_result":
        tr = event.get("tool_result") or {}
        return frame("tool_result", name=tr.get("name", ""), status=tr.get("status", ""),
                     bytes=_content_bytes(tr.get("content")))
    return None


def estimate_usd(usage: dict, model: str) -> float:
    family = next((k for k in PRICES if k in model), "sonnet")
    inp, out, cache = PRICES[family]
    return round(
        usage.get("inputTokens", 0) / 1e6 * inp
        + usage.get("outputTokens", 0) / 1e6 * out
        + usage.get("cacheReadInputTokens", 0) / 1e6 * cache,
        4,
    )


def metrics_frame(result: Any, model: str) -> dict:
    usage = dict(getattr(result.metrics, "accumulated_usage", {}) or {})
    return frame(
        "metrics",
        input_tokens=usage.get("inputTokens", 0),
        output_tokens=usage.get("outputTokens", 0),
        cache_read_tokens=usage.get("cacheReadInputTokens", 0),
        cycles=getattr(result.metrics, "cycle_count", 0),
        stop_reason=str(getattr(result, "stop_reason", "")),
        est_usd=estimate_usd(usage, model),
    )
```

```python
"""Startup self-checks logged to CloudWatch (spec 5). Hard failures keep /ping unhealthy; warnings never crash."""

from __future__ import annotations

import logging
import os
import tempfile
from collections.abc import Mapping
from dataclasses import dataclass, field
from pathlib import Path

from ops_investigator import platform

log = logging.getLogger("ops_investigator.preflight")


@dataclass
class PreflightReport:
    ok: bool = True
    checks: list[tuple[str, str, str]] = field(default_factory=list)

    def add(self, name: str, status: str, detail: str) -> None:
        self.checks.append((name, status, detail))
        if status == "FAIL":
            self.ok = False
        getattr(log, {"PASS": "info", "WARN": "warning", "FAIL": "error"}[status])("preflight %s=%s %s", name, status, detail)


def converse_probe(model_id: str, region: str) -> str | None:
    """1-token Converse call: the only reliable model-access check (P25). Returns error code or None."""
    import boto3
    from botocore.exceptions import BotoCoreError, ClientError

    try:
        boto3.client("bedrock-runtime", region_name=region).converse(
            modelId=model_id, messages=[{"role": "user", "content": [{"text": "ping"}]}],
            inferenceConfig={"maxTokens": 1})
        return None
    except ClientError as e:
        return e.response.get("Error", {}).get("Code", "ClientError")
    except BotoCoreError as e:
        return type(e).__name__


def _bare_model_id(model: str) -> str:
    return model.split("/", 1)[1] if "/" in model else model


def run_preflight(env: Mapping[str, str] | None = None, *, probe_model: bool = True) -> PreflightReport:
    env = os.environ if env is None else env
    rep = PreflightReport()
    try:
        rt = platform.detect("preflight-" + "0" * 30, "preflight", env=env)
    except platform.RuntimeConfigError as e:
        rep.add("region" if "AWS_REGION" in str(e) else "config", "FAIL", str(e))
        return rep
    rep.add("profile", "PASS", rt.profile)
    if rt.profile == "local":
        return rep
    tmp = Path(tempfile.gettempdir())
    try:
        probe = tmp / ".preflight-write"
        probe.write_text("ok")
        probe.unlink()
        rep.add("tempdir", "PASS", str(tmp))
    except OSError as e:
        rep.add("tempdir", "FAIL", f"{tmp} is not writable: {e}. The harness needs a writable workspace (P2).")
    rep.add("memory_id", "PASS", rt.memory_id or "")
    if env.get("OTEL_TRACES_EXPORTER"):
        rep.add("otel_exporter", "WARN", "OTEL_TRACES_EXPORTER is set; ADOT should own the exporter in AgentCore. Unset it (P8).")
    else:
        rep.add("otel_exporter", "PASS", "unset; ADOT owns tracing")
    if probe_model:
        err = converse_probe(_bare_model_id(rt.model), rt.region or "")
        if err:
            rep.add("model", "WARN", f"{rt.model}: {err}. Enable model access in Bedrock console or fix the execution role.")
        else:
            rep.add("model", "PASS", rt.model)
    return rep
```

```python
"""AgentCore HTTP entrypoint (spec 4.4, 5). Streams versioned SSE frames; never raw Strands events."""

from __future__ import annotations

import asyncio
import contextlib
import logging
import os
from collections import OrderedDict
from collections.abc import Callable
from typing import Any

from bedrock_agentcore.runtime import BedrockAgentCoreApp

from ops_investigator import platform, sse
from ops_investigator.agent import LIMITS, build_agent
from ops_investigator.auth import ActorError, actor_from_headers
from ops_investigator.preflight import run_preflight

log = logging.getLogger("ops_investigator.http")

MAX_PROMPT_BYTES = 32_768
MAX_SECONDS = float(os.environ.get("HARNESS_MAX_SECONDS", "600"))
MIN_SESSION_LEN = 33


class PayloadError(ValueError):
    pass


class ActorMismatch(PermissionError):
    pass


def validate_payload(payload: Any) -> dict:
    if not isinstance(payload, dict):
        raise PayloadError("payload must be a JSON object with a string 'prompt'")
    unknown = set(payload) - {"prompt", "actor_id"}
    if unknown:
        raise PayloadError(f"unknown payload keys: {sorted(unknown)}")
    prompt = payload.get("prompt")
    if not isinstance(prompt, str) or not prompt.strip():
        raise PayloadError("'prompt' must be a non-empty string")
    if len(prompt.encode()) > MAX_PROMPT_BYTES:
        raise PayloadError(f"'prompt' exceeds {MAX_PROMPT_BYTES} bytes")
    actor = payload.get("actor_id")
    if actor is not None and not isinstance(actor, str):
        raise PayloadError("'actor_id' must be a string when present")
    return {"prompt": prompt, "actor_id": actor}


class AgentCache:
    """Small LRU keyed by session id; a session belongs to exactly one actor (spec 5)."""

    def __init__(self, maxsize: int = 16) -> None:
        self.maxsize = maxsize
        self._items: OrderedDict[str, tuple[str, Any, asyncio.Lock]] = OrderedDict()

    def __contains__(self, session_id: str) -> bool:
        return session_id in self._items

    def clear(self) -> None:
        self._items.clear()

    def get_or_create(self, session_id: str, actor_id: str, factory: Callable[[], Any]) -> Any:
        if session_id in self._items:
            actor, agent, _ = self._items[session_id]
            if actor != actor_id:
                raise ActorMismatch("this session belongs to a different actor")
            self._items.move_to_end(session_id)
            return agent
        agent = factory()
        self._items[session_id] = (actor_id, agent, asyncio.Lock())
        while len(self._items) > self.maxsize:
            _, (_, old, _) = self._items.popitem(last=False)
            _close(old)
        return agent

    def lock(self, session_id: str) -> asyncio.Lock:
        return self._items[session_id][2]


def _close(agent: Any) -> None:
    sm = getattr(agent, "session_manager", None)
    if sm is not None and hasattr(sm, "close"):
        with contextlib.suppress(Exception):
            sm.close()


CACHE = AgentCache(maxsize=16)
app = BedrockAgentCoreApp()
_PROFILE = platform.detect("startup-" + "0" * 26, "startup").profile if os.environ.get(platform.MEMORY_ENV) else "local"


async def _stream(agent: Any, prompt: str, limits: dict, model: str):
    result = None
    async for event in agent.stream_async(prompt, limits=limits):
        if isinstance(event, dict) and "result" in event:
            result = event["result"]
            continue
        frame = sse.map_event(event)
        if frame:
            yield frame
    if result is not None:
        yield sse.metrics_frame(result, model)
        stop = str(getattr(result, "stop_reason", ""))
        if stop.startswith("limit_"):
            yield sse.frame("error", message=f"budget exhausted ({stop})",
                            hint="the agent hit its turn/token limit; asking it to summarise what it found")
            closing = dict(limits, turns=1)
            async for event in agent.stream_async(
                "You hit the budget. Summarise findings so far, likely cause, and what remains unchecked.", limits=closing
            ):
                frame = sse.map_event(event)
                if frame:
                    yield frame


@app.entrypoint
async def invoke(payload: Any, context: Any):
    session_id = getattr(context, "session_id", None) or ""
    headers = getattr(context, "request_headers", None) or {}
    if len(session_id) < MIN_SESSION_LEN:
        yield sse.frame("error", message="runtimeSessionId too short",
                        hint=f"AgentCore requires >= {MIN_SESSION_LEN} characters; use scripts/invoke.py which generates one (P9)")
        return
    try:
        body = validate_payload(payload)
        actor = actor_from_headers(headers, profile=_PROFILE, payload_actor=body["actor_id"])
    except (PayloadError, ActorError) as e:
        yield sse.frame("error", message=str(e), hint="fix the request and retry")
        return
    try:
        agent = CACHE.get_or_create(session_id, actor, lambda: build_agent(session_id, actor))
    except ActorMismatch as e:
        yield sse.frame("error", message=str(e), hint="start a new session id for this actor")
        return
    except Exception as e:  # session manager init (memory AccessDenied etc.) fails closed (spec 5)
        log.exception("agent construction failed")
        yield sse.frame("error", message=f"agent unavailable: {type(e).__name__}",
                        hint="check preflight lines in the runtime logs (agentcore logs)")
        return
    lock = CACHE.lock(session_id)
    if lock.locked():
        yield sse.frame("error", message="busy", hint="a turn is already running on this session; wait for done")
        return
    model = platform.detect(session_id, actor).model
    async with lock:
        try:
            async with asyncio.timeout(MAX_SECONDS):
                async for frame in _stream(agent, body["prompt"], LIMITS, model):
                    yield frame
        except TimeoutError:
            yield sse.frame("error", message=f"timeout after {MAX_SECONDS:.0f}s", hint="narrow the question or raise HARNESS_MAX_SECONDS")
        except Exception as e:
            log.exception("turn failed")
            yield sse.frame("error", message=f"{type(e).__name__}: {e}"[:300], hint="see runtime logs")
    yield sse.frame("done")


def startup() -> None:
    rep = run_preflight()
    if not rep.ok:
        from bedrock_agentcore.runtime.models import PingStatus

        log.error("preflight failed; invocations will return errors until fixed")
        app.force_ping_status(PingStatus.HEALTHY_BUSY)
```

```python
"""A2A 1.0 agent card, hand-built on a2a-sdk 1.x protobuf types (spec 4.5, P22).

Strands' own A2AServer pins a2a-sdk 0.3 and emits a 0.3 card; this runtime speaks 1.0 end to end.

from __future__ import annotations

from google.protobuf.json_format import MessageToDict

from a2a.types import (
    AgentCapabilities,
    AgentCard,
    AgentInterface,
    AgentProvider,
    AgentSkill,
    OpenIdConnectSecurityScheme,
    SecurityRequirement,
    SecurityScheme,
)

NAME = "AWS Ops Investigator"
DESCRIPTION = ("Read-only diagnostician for one AWS account: failures, changes, and cost, with evidence. "
               "Built on the Strands harness and hosted on Amazon Bedrock AgentCore Runtime.")

SKILLS = [
    AgentSkill(id="investigate-incident", name="Investigate an incident",
               description="Why is this Lambda/ECS service/API failing or slow? Evidence from config, CloudTrail, and logs.",
               tags=["aws", "ops", "incident"], examples=["Why did orders-api start returning 5XX an hour ago?"],
               input_modes=["text/plain"], output_modes=["text/plain"]),
    AgentSkill(id="cost-review", name="Cost review",
               description="Where is spend going and what changed? Cost Explorer by service, region, or usage type.",
               tags=["aws", "cost"], examples=["What cost the most in the last 7 days and why?"],
               input_modes=["text/plain"], output_modes=["text/plain"]),
    AgentSkill(id="inventory", name="Inventory a resource or region",
               description="List and summarise resources (read-only) with ids, ARNs, and state.",
               tags=["aws", "inventory"], examples=["What AgentCore runtimes exist in us-west-2 and are they READY?"],
               input_modes=["text/plain"], output_modes=["text/plain"]),
]


def build_agent_card(*, public_url: str, discovery_url: str | None, version: str = "0.1.0") -> AgentCard:
    card = AgentCard(
        name=NAME,
        description=DESCRIPTION,
        version=version,
        provider=AgentProvider(organization="strands-harness-on-aws", url="https://github.com/iracic82/strands-harness-on-aws"),
        documentation_url="https://github.com/iracic82/strands-harness-on-aws#a2a",
        capabilities=AgentCapabilities(streaming=True, push_notifications=False, extended_agent_card=False),
        default_input_modes=["text/plain"],
        default_output_modes=["text/plain"],
        supported_interfaces=[AgentInterface(url=public_url, protocol_binding="JSONRPC", protocol_version="1.0")],
        skills=SKILLS,
    )
    if discovery_url:
        card.security_schemes["cognito"].CopyFrom(SecurityScheme(open_id_connect_security_scheme=OpenIdConnectSecurityScheme(
            open_id_connect_url=discovery_url, description="Cognito user pool JWT (AgentCore CUSTOM_JWT inbound auth)")))
        req = SecurityRequirement()
        req.schemes["cognito"]  # creates the empty scopes entry
        card.security_requirements.append(req)
    return card


def card_json(card: AgentCard) -> dict:
    return MessageToDict(card)
```

```python
import asyncio
import logging
import os

from typing import Any

import uvicorn

from a2a.server.request_handlers import DefaultRequestHandler

log = logging.getLogger("ops_investigator.a2a")
MAX_SECONDS = float(os.environ.get("HARNESS_MAX_SECONDS", "600"))


def _headers(context: RequestContext) -> dict[str, str]:
    cc = getattr(context, "call_context", None)
    state = getattr(cc, "state", None) or {}
    hdrs = state.get("headers") if isinstance(state, dict) else None
    return dict(hdrs or {})


class HarnessExecutor(AgentExecutor):
    def __init__(self, agent_factory: Callable[[str, str], Any] | None = None) -> None:
        self._factory = agent_factory or build_agent
        self._agents: dict[str, Any] = {}
        self._profile = platform.detect("a2a-startup-" + "0" * 22, "startup").profile if os.environ.get(platform.MEMORY_ENV) else "local"

    def _agent_for(self, context_id: str, actor: str) -> Any:
        key = f"{context_id}/{actor}"
        if key not in self._agents:
            # AgentCore session ids must be >= 33 chars; A2A context ids are uuid4 (36) so they qualify (P9).
            self._agents[key] = self._factory(context_id, actor)
        return self._agents[key]

    async def execute(self, context: RequestContext, event_queue: EventQueue) -> None:
        task = context.current_task or new_task_from_user_message(context.message)
        if not context.current_task:
            await event_queue.enqueue_event(task)
        updater = TaskUpdater(event_queue=event_queue, task_id=task.id, context_id=task.context_id)
        try:
            actor = actor_from_headers(_headers(context), profile=self._profile, payload_actor=None)
        except ActorError as e:
            await updater.update_status(state=TaskState.TASK_STATE_REJECTED, message=new_text_message(str(e)))
            return
        await updater.update_status(state=TaskState.TASK_STATE_WORKING, message=new_text_message("Investigating..."))
        prompt = get_message_text(context.message) or ""
        agent = self._agent_for(task.context_id, actor)
        chunks: list[str] = []
        try:
            async with asyncio.timeout(MAX_SECONDS):
                async for ev in agent.stream_async(prompt, limits=LIMITS):
                    if isinstance(ev, dict) and isinstance(ev.get("data"), str):
                        chunks.append(ev["data"])
        except TimeoutError:
            await updater.update_status(state=TaskState.TASK_STATE_FAILED, message=new_text_message(f"timeout after {MAX_SECONDS:.0f}s"))
            return
        except Exception as e:  # noqa: BLE001
            log.exception("a2a turn failed")
            await updater.update_status(state=TaskState.TASK_STATE_FAILED, message=new_text_message(f"{type(e).__name__}"))
            return
        await updater.add_artifact(parts=[new_text_part(text="".join(chunks) or "(no answer)", media_type="text/plain")],
                                   name="answer")
        await updater.update_status(state=TaskState.TASK_STATE_COMPLETED)

    async def cancel(self, context: RequestContext, event_queue: EventQueue) -> None:
        updater = TaskUpdater(event_queue=event_queue, task_id=context.task_id or "", context_id=context.context_id or "")
        await updater.update_status(state=TaskState.TASK_STATE_CANCELED)


def build_app(*, public_url: str, discovery_url: str | None, agent_factory=None) -> Starlette:
    card = build_agent_card(public_url=public_url, discovery_url=discovery_url)
    handler = DefaultRequestHandler(agent_executor=HarnessExecutor(agent_factory), task_store=InMemoryTaskStore(), agent_card=card)
    routes = [*create_agent_card_routes(card), *create_jsonrpc_routes(handler, "/")]
    app = Starlette(routes=routes)

    @app.route("/ping", methods=["GET"])
    async def ping(_request):  # AgentCore health contract applies to every protocol
        from starlette.responses import JSONResponse
        return JSONResponse({"status": "Healthy"})

    return app


def main() -> None:
    app = build_app(public_url=os.environ.get("A2A_PUBLIC_URL", "http://localhost:9000"),
                    discovery_url=os.environ.get("COGNITO_DISCOVERY_URL") or None)
    uvicorn.run(app, host="0.0.0.0", port=9000)
```

```python
#!/usr/bin/env python3
"""doctor.py: preflight for running the Strands harness on AgentCore. Drop into any harness project.

Usage: python doctor.py --region us-west-2 [--model bedrock/global.anthropic.claude-sonnet-5] [--build Container] [--json]
Exit code: 0 all PASS, 1 any WARN, 2 any FAIL. Every FAIL/WARN prints the fix. Needs only boto3.
"""

from __future__ import annotations

import argparse
import json
import platform as _platform
import shutil
import subprocess
import sys
from collections.abc import Callable
from dataclasses import asdict, dataclass

import boto3
from botocore.exceptions import BotoCoreError, ClientError, NoCredentialsError

DEFAULT_MODEL = "bedrock/global.anthropic.claude-sonnet-5"
DEFAULT_SMALL = "global.anthropic.claude-haiku-4-5-20251001-v1:0"   # the harness's hidden second model (P4)
STATUS_ORDER = {"PASS": 0, "WARN": 1, "FAIL": 2}


@dataclass
class Check:
    name: str
    status: str
    detail: str
    fix: str


def _default_clients(service: str, region: str):
    return boto3.client(service, region_name=region)


def _cmd(run_cmd, args: list[str]) -> tuple[int, str]:
    try:
        r = run_cmd(args, capture_output=True, text=True, timeout=30)
        return r.returncode, (r.stdout or "") + (r.stderr or "")
    except (OSError, subprocess.SubprocessError) as e:
        return 127, str(e)


def _converse(clients, region: str, model_id: str) -> str | None:
    try:
        clients("bedrock-runtime", region).converse(modelId=model_id, messages=[{"role": "user", "content": [{"text": "ping"}]}],
                                                    inferenceConfig={"maxTokens": 1})
        return None
    except ClientError as e:
        return e.response.get("Error", {}).get("Code", "ClientError")
    except (BotoCoreError, NoCredentialsError) as e:
        return type(e).__name__


def run(region: str, model: str, small_model: str, build: str = "Container", *, clients=None,
        which: Callable[[str], str | None] = shutil.which, run_cmd=subprocess.run) -> list[Check]:
    clients = clients or _default_clients
    out: list[Check] = []

    v = sys.version_info
    out.append(Check("python", "PASS" if v >= (3, 10) else "FAIL", f"{v.major}.{v.minor}", "Install Python 3.10+ (3.12 recommended)."))
    out.append(Check("uv", "PASS" if which("uv") else "WARN", which("uv") or "missing", "curl -LsSf https://astral.sh/uv/install.sh | sh"))
    out.append(Check("node", "PASS" if which("node") else "FAIL", which("node") or "missing", "Install Node 20+ (needed for the @aws/agentcore CLI and CDK)."))

    code, text = _cmd(run_cmd, ["agentcore", "--version"])
    if code != 0:
        out.append(Check("agentcore_cli", "FAIL", "agentcore not found", "npm install -g @aws/agentcore"))
    elif "Usage: agentcore [OPTIONS]" in text:  # the deprecated Python starter toolkit prints Click-style usage (P1/P20)
        out.append(Check("agentcore_cli", "FAIL", "deprecated Python starter toolkit is first on PATH",
                         "pip uninstall bedrock-agentcore-starter-toolkit; npm install -g @aws/agentcore"))
    else:
        out.append(Check("agentcore_cli", "PASS", text.strip().splitlines()[0][:40], ""))

    try:
        ident = clients("sts", region).get_caller_identity()
        out.append(Check("credentials", "PASS", f"{ident['Account']} {ident['Arn'].rsplit('/', 1)[-1]}", ""))
    except (ClientError, BotoCoreError, NoCredentialsError) as e:
        out.append(Check("credentials", "FAIL", type(e).__name__, "aws sso login --profile <profile>; export AWS_PROFILE=<profile>"))
        return out
    out.append(Check("region_set", "PASS" if region else "FAIL", region or "unset", "export AWS_REGION=us-west-2 (P5)"))

    try:
        clients("bedrock-agentcore-control", region).list_agent_runtimes(maxResults=1)
        out.append(Check("agentcore_region", "PASS", region, ""))
    except (ClientError, BotoCoreError) as e:
        out.append(Check("agentcore_region", "FAIL", f"{type(e).__name__}", f"AgentCore Runtime is not reachable in {region}; pick a supported region (P7)."))

    bare = model.split("/", 1)[-1]
    for name, mid in (("model_main", bare), ("model_small", small_model)):
        err = _converse(clients, region, mid)
        out.append(Check(name, "PASS" if err is None else "FAIL", mid if err is None else f"{mid}: {err}",
                         "Bedrock console > Model access: enable the model (and its global/us inference profile); the harness uses Haiku 4.5 for web_fetch/memory (P4, P25)."))

    try:
        dest = clients("xray", region).get_trace_segment_destination()
        active = dest.get("Status") == "ACTIVE" and dest.get("Destination") == "CloudWatchLogs"
        out.append(Check("transaction_search", "PASS" if active else "FAIL", json.dumps(dest, default=str)[:80],
                         "Enable CloudWatch Transaction Search once per region: aws xray update-trace-segment-destination --destination CloudWatchLogs (P8)"))
    except (ClientError, BotoCoreError) as e:
        out.append(Check("transaction_search", "WARN", type(e).__name__, "Could not query X-Ray; verify Transaction Search in the CloudWatch console (P8)."))

    try:
        clients("cloudformation", region).describe_stacks(StackName="CDKToolkit")
        out.append(Check("cdk_bootstrap", "PASS", "CDKToolkit present", ""))
    except (ClientError, BotoCoreError):
        out.append(Check("cdk_bootstrap", "WARN", "CDKToolkit stack missing", f"npx cdk bootstrap aws://<account>/{region} (agentcore deploy runs it if allowed)"))

    if build == "Container":
        code, text = _cmd(run_cmd, ["docker", "buildx", "inspect", "--bootstrap"])
        ok = code == 0 and "linux/arm64" in text
        out.append(Check("docker_arm64", "PASS" if ok else "FAIL", "linux/arm64 available" if ok else text.strip()[-80:],
                         "AgentCore needs arm64 images. Install Docker with buildx and QEMU (docker run --privileged tonistiigi/binfmt --install arm64) (P16)."))

    code, text = _cmd(run_cmd, ["git", "ls-files", ".agent", "app/ops_investigator/.agent"])
    out.append(Check("agent_dir_untracked", "PASS" if not text.strip() else "FAIL", "clean" if not text.strip() else text.strip()[:80],
                     "Local sessions/memory are plaintext transcripts: git rm -r --cached .agent && add .agent/ to .gitignore"))
    return out


def worst(checks: list[Check]) -> int:
    return max((STATUS_ORDER[c.status] for c in checks), default=0)


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--region", default=None)
    p.add_argument("--model", default=DEFAULT_MODEL)
    p.add_argument("--small-model", default=DEFAULT_SMALL)
    p.add_argument("--build", default="Container", choices=["Container", "CodeZip"])
    p.add_argument("--json", action="store_true")
    a = p.parse_args(argv)
    region = a.region or boto3.session.Session().region_name or ""
    checks = run(region, a.model, a.small_model, a.build)
    if a.json:
        print(json.dumps([asdict(c) for c in checks], indent=2))
    else:
        for c in checks:
            line = f"[{c.status:4}] {c.name:20} {c.detail}"
            print(line if c.status == "PASS" else f"{line}\n       fix: {c.fix}")
    return worst(checks)


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""Create (idempotently) the Cognito user pool + app client used for AgentCore CUSTOM_JWT inbound auth (spec 2, 4.4)."""

from __future__ import annotations

import argparse
import json
import os
import re
from pathlib import Path

import boto3

ROOT = Path(__file__).resolve().parents[1]


def find_pool(idp, name: str) -> str | None:
    for page in idp.get_paginator("list_user_pools").paginate(MaxResults=60):
        for p in page["UserPools"]:
            if p["Name"] == name:
                return p["Id"]
    return None


def find_client(idp, pool_id: str, name: str) -> str | None:
    for page in idp.get_paginator("list_user_pool_clients").paginate(UserPoolId=pool_id, MaxResults=60):
        for c in page["UserPoolClients"]:
            if c["ClientName"] == name:
                return c["ClientId"]
    return None


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--region", required=True)
    ap.add_argument("--project", default="harnessaws")
    ap.add_argument("--username", default="demo")
    ap.add_argument("--password", default=os.environ.get("DEMO_PASSWORD"))
    a = ap.parse_args()
    if not a.password:
        ap.error("--password or DEMO_PASSWORD is required (min 8 chars, upper/lower/digit)")
    idp = boto3.client("cognito-idp", region_name=a.region)
    pool_name, client_name = f"{a.project}-users", f"{a.project}-cli"
    pool_id = find_pool(idp, pool_name) or idp.create_user_pool(
        PoolName=pool_name, Policies={"PasswordPolicy": {"MinimumLength": 8}},
        AdminCreateUserConfig={"AllowAdminCreateUserOnly": True})["UserPool"]["Id"]
    client_id = find_client(idp, pool_id, client_name) or idp.create_user_pool_client(
        UserPoolId=pool_id, ClientName=client_name, GenerateSecret=False,
        ExplicitAuthFlows=["ALLOW_USER_PASSWORD_AUTH", "ALLOW_REFRESH_TOKEN_AUTH"])["UserPoolClient"]["ClientId"]
    try:
        idp.admin_get_user(UserPoolId=pool_id, Username=a.username)
    except idp.exceptions.UserNotFoundException:
        idp.admin_create_user(UserPoolId=pool_id, Username=a.username, MessageAction="SUPPRESS")
    idp.admin_set_user_password(UserPoolId=pool_id, Username=a.username, Password=a.password, Permanent=True)
    discovery = f"https://cognito-idp.{a.region}.amazonaws.com/{pool_id}/.well-known/openid-configuration"

    cfg_path = ROOT / "agentcore" / "agentcore.json"
    text = cfg_path.read_text()
    text = re.sub(r'"discoveryUrl": "[^"]*"', f'"discoveryUrl": "{discovery}"', text)
    text = re.sub(r'"allowedClients": \[[^\]]*\]', f'"allowedClients": ["{client_id}"]', text)
    text = re.sub(r'\{"name": "COGNITO_DISCOVERY_URL", "value": "[^"]*"\}',
                  f'{{"name": "COGNITO_DISCOVERY_URL", "value": "{discovery}"}}', text)
    cfg_path.write_text(text)
    json.loads(text)  # still valid JSON
    print(json.dumps({"user_pool_id": pool_id, "client_id": client_id, "discovery_url": discovery, "username": a.username}, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
"""Invoke the deployed HTTP runtime with a bearer token and print SSE frames (spec 4.4)."""

from __future__ import annotations

import argparse
import json
import sys
import uuid

import boto3


def new_session_id() -> str:
    return uuid.uuid4().hex + uuid.uuid4().hex[:8]   # 40 chars; AgentCore needs >= 33 (P9)


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--arn", required=True)
    ap.add_argument("--region", required=True)
    ap.add_argument("--prompt", required=True)
    ap.add_argument("--session", default=None)
    ap.add_argument("--token", default=None, help="Cognito JWT; required when the runtime uses CUSTOM_JWT")
    ap.add_argument("--raw", action="store_true")
    a = ap.parse_args()
    session = a.session or new_session_id()
    print(f"session_id={session}", file=sys.stderr)
    client = boto3.client("bedrock-agentcore", region_name=a.region)
    if a.token:
        client.meta.events.register("before-sign.bedrock-agentcore.InvokeAgentRuntime",
                                    lambda request, **kw: request.headers.__setitem__("Authorization", f"Bearer {a.token}"))
    resp = client.invoke_agent_runtime(agentRuntimeArn=a.arn, runtimeSessionId=session,
                                       payload=json.dumps({"prompt": a.prompt}).encode())
    body = resp["response"]
    for line in body.iter_lines():
        if not line:
            continue
        text = line.decode() if isinstance(line, bytes) else line
        if text.startswith("data: "):
            text = text[6:]
        if a.raw:
            print(text)
            continue
        try:
            frame = json.loads(text)
        except ValueError:
            print(text)
            continue
        t = frame.get("type")
        if t == "text":
            print(frame["delta"], end="", flush=True)
        elif t in ("tool_use", "denied", "error", "metrics"):
            print(f"\n[{t}] {json.dumps({k: v for k, v in frame.items() if k not in ('v', 'type')})}", flush=True)
        elif t == "done":
            print()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

Note: for CUSTOM_JWT runtimes the data-plane call is a plain HTTPS POST to
`https://bedrock-agentcore.<region>.amazonaws.com/runtimes/<urlencoded ARN>/invocations?qualifier=DEFAULT` with
`Authorization: Bearer` and `X-Amzn-Bedrock-AgentCore-Runtime-Session-Id`; if the boto3 SigV4 path is rejected for a
JWT runtime, implement that request with `httpx` instead (spec 10.1 verifies which works and PITFALLS records it).

```python
#!/usr/bin/env python3
"""Fetch the A2A agent card and send one message (a2a-sdk 1.x). Warns if the card is 0.3-shaped (P22)."""

from __future__ import annotations

import argparse
import asyncio
import json

import httpx
from a2a.client import ClientConfig, create_client
from a2a.helpers import new_text_message
from a2a.types import Role, SendMessageRequest


async def run(url: str, token: str | None, prompt: str) -> int:
    headers = {"Authorization": f"Bearer {token}"} if token else {}
    async with httpx.AsyncClient(headers=headers, timeout=120) as http:
        card = (await http.get(url.rstrip("/") + "/.well-known/agent-card.json")).json()
        print(json.dumps({k: card.get(k) for k in ("name", "version", "supportedInterfaces", "securitySchemes")}, indent=2))
        version = (card.get("supportedInterfaces") or [{}])[0].get("protocolVersion") or card.get("protocolVersion")
        if version != "1.0":
            print(f"WARNING: card advertises protocolVersion={version!r}; this client speaks 1.0 (see PITFALLS P22)")
        client = await create_client(url, ClientConfig(streaming=False, httpx_client=http))
        resp = await client.send_message(SendMessageRequest(message=new_text_message(prompt, role=Role.ROLE_USER)))
        print(resp)
        await client.close()
    return 0


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--url", required=True)
    ap.add_argument("--token", default=None)
    ap.add_argument("--prompt", default="What AgentCore runtimes exist in this region and are they READY?")
    a = ap.parse_args()
    return asyncio.run(run(a.url, a.token, a.prompt))


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
"""Delete one actor's long-term records and events from AgentCore Memory (admin escape hatch, spec 4.3)."""

from __future__ import annotations

import argparse

import boto3


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--region", required=True)
    ap.add_argument("--memory-id", required=True)
    ap.add_argument("--actor", required=True)
    a = ap.parse_args()
    dp = boto3.client("bedrock-agentcore", region_name=a.region)
    deleted = 0
    for ns in (f"/preferences/{a.actor}/", f"/facts/{a.actor}/"):
        for page in dp.get_paginator("list_memory_records").paginate(memoryId=a.memory_id, namespace=ns):
            for rec in page.get("memoryRecordSummaries", []):
                dp.delete_memory_record(memoryId=a.memory_id, memoryRecordId=rec["memoryRecordId"])
                deleted += 1
    for page in dp.get_paginator("list_sessions").paginate(memoryId=a.memory_id, actorId=a.actor):
        for s in page.get("sessionSummaries", []):
            for ev in dp.get_paginator("list_events").paginate(memoryId=a.memory_id, actorId=a.actor, sessionId=s["sessionId"]):
                for e in ev.get("events", []):
                    dp.delete_event(memoryId=a.memory_id, actorId=a.actor, sessionId=s["sessionId"], eventId=e["eventId"])
                    deleted += 1
    print(f"deleted {deleted} records/events for actor {a.actor}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

"""Run every ```bash block in README.md and docs/*.md that is not marked title="manual"; validate Mermaid blocks."""

from __future__ import annotations

import re
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
FENCE = re.compile(r"```(\w+)([^\n]*)\n(.*?)```", re.S)


def blocks(path: Path):
    for lang, attrs, body in FENCE.findall(path.read_text()):
        yield lang, attrs.strip(), body


def main() -> int:
    failures = 0
    for md in [ROOT / "README.md", *sorted((ROOT / "docs").glob("*.md"))]:
        for lang, attrs, body in blocks(md):
            if lang == "bash" and 'title="manual"' not in attrs:

                print(f"[{status}] {md.name}: {body.strip().splitlines()[0][:70]}")
                if r.returncode:
                    failures += 1
                    print(r.stdout[-800:], r.stderr[-800:], sep="\n")

                print(f"[{'ok' if r.returncode == 0 else 'FAIL'}] {md.name}: mermaid block")

    return 1 if failures else 0


if __name__ == "__main__":

```python
#!/usr/bin/env python3
"""One-time local setup: detect account/region, write agentcore/aws-targets.json and .env. Idempotent."""

from __future__ import annotations

import argparse
import json
import os
import re
import sys
from pathlib import Path

import boto3

ROOT = Path(__file__).resolve().parents[1]


def detect_region(explicit: str | None) -> str:
    return explicit or os.environ.get("AWS_REGION") or boto3.session.Session().region_name or "us-west-2"


def write_if_absent(path: Path, content: str, force: bool) -> str:
    if path.exists() and not force:
        return "kept" if path.read_text() == content else "exists (differs; use --force to overwrite)"
    path.write_text(content)
    return "written"


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--region", default=None)
    ap.add_argument("--force", action="store_true")
    ap.add_argument("--reset-placeholders", action="store_true", help="restore REPLACED_BY_* placeholders in agentcore.json")
    a = ap.parse_args()
    region = detect_region(a.region)
    ident = boto3.client("sts", region_name=region).get_caller_identity()
    account = ident["Account"]
    targets = json.loads((ROOT / "agentcore" / "aws-targets.example.json").read_text())
    for t in targets:
        t["account"], t["region"] = account, region
    print("aws-targets.json:", write_if_absent(ROOT / "agentcore" / "aws-targets.json", json.dumps(targets, indent=2) + "\n", a.force))
    env = (ROOT / ".env.example").read_text().replace("<region>", region).replace("<account-id>", account)
    print(".env:", write_if_absent(ROOT / ".env", env, a.force))
    if a.reset_placeholders:
        p = ROOT / "agentcore" / "agentcore.json"
        text = p.read_text()
        text = re.sub(r'"discoveryUrl": "[^"]*"', '"discoveryUrl": "REPLACED_BY_auth_bootstrap"', text)
        text = re.sub(r'"allowedClients": \[[^\]]*\]', '"allowedClients": ["REPLACED_BY_auth_bootstrap"]', text)
        text = re.sub(r'\{"name": "COGNITO_DISCOVERY_URL", "value": "[^"]*"\}', '{"name": "COGNITO_DISCOVERY_URL", "value": "REPLACED_BY_auth_bootstrap"}', text)
        text = re.sub(r'\{"name": "A2A_PUBLIC_URL", "value": "[^"]*"\}', '{"name": "A2A_PUBLIC_URL", "value": "REPLACED_AFTER_FIRST_DEPLOY"}', text)
        p.write_text(text)
        print("agentcore.json: placeholders restored")
    print(f"\naccount={account} region={region} caller={ident['Arn'].rsplit('/', 1)[-1]}\nnext: make doctor")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
